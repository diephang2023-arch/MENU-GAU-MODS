local function mapText(s)
	if not ENABLE_CUSTOM_OVERLAY or type(s) ~= "string" or s == "" then return s end
	return OVERLAY_CONFIG.TEXT[s] or OVERLAY_CONFIG.IDS[s] or OVERLAY_CONFIG.LINKS[s] or s
end

local function mapImage(img)
	if not ENABLE_CUSTOM_OVERLAY or type(img) ~= "string" or img == "" then return img end
	local direct = OVERLAY_CONFIG.IMAGES[img] or OVERLAY_CONFIG.IDS[img]
	if direct ~= nil then return tostring(direct) end
	local digits = img:match("(%d+)")
	if digits then
		local rep = OVERLAY_CONFIG.IMAGES[digits] or OVERLAY_CONFIG.IDS[digits]
		if rep ~= nil then
			rep = tostring(rep)
			return rep:match("^rbx") and rep or ("rbxassetid://" .. rep)
		end
	end
	return img
end

local function mapClipboard(txt)
	if not ENABLE_CUSTOM_OVERLAY or type(txt) ~= "string" or txt == "" then return txt end
	return OVERLAY_CONFIG.CLIPBOARD[txt] or OVERLAY_CONFIG.IDS[txt] or OVERLAY_CONFIG.LINKS[txt] or txt
end

local function patchGuiOnce(root)
	if not root then return end
	pcall(function()
		for _, obj in ipairs(root:GetDescendants()) do
			if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
				local m = mapText(obj.Text)
				if m ~= obj.Text then obj.Text = m end
			elseif obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
				local m = mapImage(obj.Image)
				if m ~= obj.Image then obj.Image = m end
			end
		end
	end)
end

-- Quét nhẹ 1 lượt lúc execute và theo dõi nhẹ khi mở tab/menu mới
local function attachLightWatcher(root)
	if not root then return end
	patchGuiOnce(root)
	pcall(function()
		root.DescendantAdded:Connect(function(obj)
			if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
				local m = mapText(obj.Text)
				if m ~= obj.Text then obj.Text = m end
			elseif obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
				local m = mapImage(obj.Image)
				if m ~= obj.Image then obj.Image = m end
			end
		end)
	end)
end

if gethui then pcall(function() attachLightWatcher(gethui()) end) end
attachLightWatcher(CoreGui)
if not PlayerGui and LocalPlayer then PlayerGui = LocalPlayer:FindFirstChildOfClass("PlayerGui") end
if PlayerGui then attachLightWatcher(PlayerGui) end
