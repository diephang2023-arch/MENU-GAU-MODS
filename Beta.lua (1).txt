local UUID_RE = "(%x%x%x%x%x%x%x%x%-%x%x%x%x%-%x%x%x%x%-%x%x%x%x%-%x%x%x%x%x%x%x%x%x%x%x%x)"
local function extractCFJobIds(body)
    if type(body) ~= "string" then return end
    local ids = {}
    for uuid in body:gmatch(UUID_RE) do
        if uuid ~= game.JobId then
            local dup = false
            for _, existing in ipairs(ids) do if existing == uuid then dup=true; break end end
            if not dup then table.insert(ids, uuid) end
        end
    end
    if #ids >= 1 then
        _G.__CF_JobIds      = ids
        _G.__CF_JobIds_tick = tick()
    end
end

local function fitActionRowButtons(container)
    if not (container and container:IsA("GuiObject")) then return end

    local visibleButtons = {}
    for _, child in ipairs(container:GetChildren()) do
        if child:IsA("GuiObject") and child.Visible then
            table.insert(visibleButtons, child)
        end
    end
    if #visibleButtons == 0 then return end

    table.sort(visibleButtons, function(a, b)
        local pa, pb = a.AbsolutePosition, b.AbsolutePosition
        if math.abs(pa.Y - pb.Y) > 12 then return pa.Y < pb.Y end
        return pa.X < pb.X
    end)

    local padPx = 6
    local listLayout = container:FindFirstChildOfClass("UIGridLayout") or container:FindFirstChildOfClass("UIListLayout")
    if listLayout then
        if listLayout:IsA("UIListLayout") then
            listLayout.FillDirection = Enum.FillDirection.Horizontal
            listLayout.SortOrder = Enum.SortOrder.LayoutOrder
            listLayout.Padding = UDim.new(0, padPx)
        elseif listLayout:IsA("UIGridLayout") then
            listLayout.SortOrder = Enum.SortOrder.LayoutOrder
            listLayout.CellPadding = UDim2.new(0, padPx, 0, 0)
        end
    end

    local totalW = container.AbsoluteSize.X
    if totalW <= 10 and container.Parent and container.Parent:IsA("GuiObject") then
        totalW = container.Parent.AbsoluteSize.X
    end
    if totalW <= 10 then return end

    local count = #visibleButtons
    local usableW = math.max(40, totalW - padPx * (count - 1))
    local eachW = math.floor(usableW / count)

    for idx, btn in ipairs(visibleButtons) do
        btn.LayoutOrder = idx
        if listLayout and listLayout:IsA("UIGridLayout") then
            listLayout.CellSize = UDim2.new(0, eachW, 1, 0)
        else
            local curH = btn.Size.Y
            if curH.Scale == 0 and curH.Offset == 0 then
                curH = UDim2.new(1, 0, 1, 0)
            end
            btn.Size = UDim2.new(0, eachW, curH.Scale, curH.Offset)
            if not listLayout then

local function removeFifthActionButtonAndFit(gui)
    if not (gui and gui:IsA("ScreenGui")) then return end

    local actionKeywords = {
        "rejoin", "server hop", "copy job id", "unload hub", "chilli hub"
    }

    local function matchesActionButton(obj)
        if not (obj:IsA("GuiButton") or obj:IsA("Frame")) then return false end
        local n = obj.Name:lower()
        for _, kw in ipairs(actionKeywords) do
            if n:find(kw, 1, true) then return true end
        end
        for _, d in ipairs(obj:GetDescendants()) do
            if d:IsA("TextLabel") or d:IsA("TextButton") then
                local t = (d.Text or ""):lower():gsub("%s+", " "):match("^%s*(.-)%s*$") or ""
                for _, kw in ipairs(actionKeywords) do
                    if t == kw or t:find(kw, 1, true) then return true end
                end
            end
        end
        return false
    end

    local containersToFit = {}
    local seenContainers = {}

    for _, d in ipairs(gui:GetDescendants()) do
        if d:IsA("GuiButton") then
            local rawTxt = ""
            if d:IsA("TextButton") then rawTxt = d.Text or "" end
            for _, sub in ipairs(d:GetDescendants()) do
                if sub:IsA("TextLabel") or sub:IsA("TextButton") then
                    rawTxt = rawTxt .. " " .. (sub.Text or "")
                end
            end
            local norm = rawTxt:lower():gsub("%s+", " "):match("^%s*(.-)%s*$") or ""

            if norm == "chilli hub" or norm:find("chilli hub", 1, true) then
                local parent = d.Parent
                if parent and parent:IsA("GuiObject") and not seenContainers[parent] then
                    seenContainers[parent] = true
                    table.insert(containersToFit, parent)
                end
                d.Visible = false
                d.Active = false
                pcall(function() d:Destroy() end)
            elseif matchesActionButton(d) then
                local parent = d.Parent
                if parent and parent:IsA("GuiObject") and not seenContainers[parent] then
                    seenContainers[parent] = true
                    table.insert(containersToFit, parent)
                end
            end
        end
    end

    for _, c in ipairs(containersToFit) do
        if c and c.Parent then
            fitActionRowButtons(c)
            if not c:GetAttribute("__ActionRowResized") then
                c:SetAttribute("__ActionRowResized", true)
                c:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
                    task.defer(function()
                        if c and c.Parent then fitActionRowButtons(c) end
                    end)
                end)
            end
        end
    end
end

local function destroyGuiNode(obj)
    if not (obj and obj.Parent and obj:IsA("GuiObject")) then return end
    pcall(function() obj.Visible = false end)
    if obj:IsA("GuiButton") then
        pcall(function() obj.Active = false end)
    end
    for _, d in ipairs(obj:GetDescendants()) do
        if d:IsA("GuiObject") then
            pcall(function() d.Visible = false end)
            if d:IsA("GuiButton") then pcall(function() d.Active = false end) end
        end
    end
    pcall(function() obj:Destroy() end)
end

local function isChilliFinderPopupMarker(str)
    local s = (str or ""):lower():gsub("%s+", " "):match("^%s*(.-)%s*$") or ""
    if s == "" then return false end
    return s:find("chilli finder", 1, true) ~= nil
        or s:find("find the best eggs", 1, true) ~= nil
        or s:find("book a slot", 1, true) ~= nil
        or s:find("open finder", 1, true) ~= nil
        or s:find("closes in ", 1, true) ~= nil
        or (s:find("ultra", 1, true) ~= nil and s:find("pro", 1, true) ~= nil and s:find("plus", 1, true) ~= nil)
end

local function removeFloatingFinder(gui)
    if not (gui and gui:IsA("ScreenGui")) then return end
    if gui.Name == "RobloxGui" or gui.Name == "TopBarApp" then return end

    local hasPopupMarker = false
    for _, d in ipairs(gui:GetDescendants()) do
        if d:IsA("TextLabel") or d:IsA("TextButton") then
            if isChilliFinderPopupMarker(d.Text) then
                hasPopupMarker = true
                local modalFrame = nil
                local cur = d.Parent
                while cur and cur ~= gui and cur:IsA("GuiObject") do
                    local sz = cur.AbsoluteSize
                    if sz.X >= 220 and sz.Y >= 140 then
                        modalFrame = cur
                    end
                    cur = cur.Parent
                end
                if modalFrame then
                    destroyGuiNode(modalFrame)
                end
            end
        end
    end

    if hasPopupMarker then
        local remainingVisible = 0
        for _, child in ipairs(gui:GetChildren()) do
            if child:IsA("GuiObject") and child.Visible then
                remainingVisible = remainingVisible + 1
            end
        end
        if remainingVisible == 0 then
            pcall(function() gui.Enabled = false end)
            pcall(function() gui:Destroy() end)
            return
        end
    end

    for _, d in ipairs(gui:GetDescendants()) do
        if d:IsA("TextLabel") or d:IsA("TextButton") then
            local raw = (d.Text or ""):lower():gsub("%s+", " "):match("^%s*(.-)%s*$") or ""
            if raw == "finder" or raw:find("^finder%s") or raw:find("%sfinder$") or raw == "chilli finder" then
                if not d:FindFirstAncestorOfClass("ScrollingFrame") then
                    local holder = d:FindFirstAncestorWhichIsA("GuiButton")
                        or d:FindFirstAncestorWhichIsA("ImageButton")
                    if not holder then
                        local cur = d.Parent
                        while cur and cur ~= gui and cur:IsA("GuiObject") do
                            local sz = cur.AbsoluteSize
                            if sz.X > 0 and sz.X <= 220 and sz.Y > 0 and sz.Y <= 90 then
                                holder = cur
                                break
                            end
                            cur = cur.Parent
                        end
                    end
                    destroyGuiNode(holder or d)
                end
            end
        elseif d:IsA("GuiObject") then
            local n = d.Name:lower()
            if n:find("finder", 1, true) and not d:FindFirstAncestorOfClass("ScrollingFrame") then
                local sz = d.AbsoluteSize
                if sz.X > 0 and sz.X <= 240 and sz.Y > 0 and sz.Y <= 110 then
                    destroyGuiNode(d)
                end
            end
        end
    end
end

                                for _, sib in ipairs(p:GetChildren()) do
                                    if sib ~= d and (sib:IsA("TextLabel") or sib:IsA("TextButton")) then
                                        local sh = sib.Text or ""
                                        if sh == "TIME LEFT" or sh == "ENDS AT"
                                        or sh == "time_left" or sh == "ends_at"
                                        or sh == "Time Left" or sh == "Ends At" then
                                            d:SetAttribute("__CFTimeLocked", true)
                                            pcall(function() d.Text = "Infinite" end)
                                            d:GetPropertyChangedSignal("Text"):Connect(function()
                                                if d.Text ~= "Infinite" then
                                                    pcall(function() d.Text = "Infinite" end)
                                                end
                                            end)
                                            break
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
                task.wait(1.0)
            end
        end)

        local pendingBatch, batchRunning = {}, false
        gui.DescendantAdded:Connect(function(newD)
            pendingBatch[#pendingBatch + 1] = newD
            if not batchRunning then
                batchRunning = true
                task.delay(0.1, function()
                    local batch  = pendingBatch
                    pendingBatch = {}
                    batchRunning = false
                    for _, d in ipairs(batch) do
                        if d and d.Parent then pcall(patchChilliElement, d) end
                    end
                end)
            end
        end)
    end

    while true do
        task.wait(1.5)
        for _, root in ipairs(roots) do
            if root then
                for _, gui in ipairs(root:GetChildren()) do
                    if gui:IsA("ScreenGui")
                    and not patchedGuis[gui]
                    and not gui:GetAttribute("__IslandLocked") then
                        task.spawn(function()
                            if isChilliFinderGui(gui) then
                                fullPatchChilliGui(gui)
                            end
                        end)
                    end
                end
            end
        end
    end
end)
