local function extractHost(u)
	local host = tostring(u or ""):match("^https?://([^/%?#]+)")
	return host and host:lower() or ""
end

local CAPTURED = {
	TEXTS = {},
	IMAGES = {},
	IDS = {},
	DISCORD = {},
	URLS = {},
	CLIPBOARD = {},
}

local SEEN = {
	TEXTS = {},
	IMAGES = {},
	IDS = {},
	DISCORD = {},
	URLS = {},
	CLIPBOARD = {},
	SOURCES = {},
}

local onStateUpdated = nil

local function isSelfContent(str)
	if type(str) ~= "string" then
		return false
	end
	return string.find(str, SELF_SIGNATURE, 1, true) ~= nil
		or (string.find(str, "GauMultiScriptScannerGui", 1, true) ~= nil and string.find(str, "AUTO_STOP_SECONDS", 1, true) ~= nil)
end

local function totalCapturedCount()
	return #CAPTURED.TEXTS + #CAPTURED.IMAGES + #CAPTURED.IDS + #CAPTURED.DISCORD + #CAPTURED.URLS + #CAPTURED.CLIPBOARD
end

local function notifyUiChanged()
	if type(onStateUpdated) == "function" then
		pcall(onStateUpdated, totalCapturedCount(), scanActive, anyScriptExecuted)
	end
end

local function resetCapturedData(keepCurrentSessionState)
	CAPTURED.TEXTS = {}
	CAPTURED.IMAGES = {}
	CAPTURED.IDS = {}
	CAPTURED.DISCORD = {}
	CAPTURED.URLS = {}
	CAPTURED.CLIPBOARD = {}

	SEEN.TEXTS = {}
	SEEN.IMAGES = {}
	SEEN.IDS = {}
	SEEN.DISCORD = {}
	SEEN.URLS = {}
	SEEN.CLIPBOARD = {}
	SEEN.SOURCES = {}

	if not keepCurrentSessionState then
		anyScriptExecuted = false
		scanActive = false
		activeHosts = {}
	end
	notifyUiChanged()
end

local function isDynamicNoiseText(str)
	if #str <= 1 then
		return true
	end
	if str:match("^[%d%s%.,:%-/%%]+$") then
		return true
	end
	local lower = str:lower()
	if lower:match("^fps%s*:") or lower:match("^ping%s*:") or lower:match("%d+%s*ms$") then
		return true
	end
	if lower:match("%d+[%.,]?%d*%s*kg") or lower:match("%d+[%.,]?%d*%s*studs") then
		return true
	end
	if str:match("^%d+:%d+$") or str:match("^%d+:%d+:%d+$") then
		return true
	end
	return false
end

local function addUniqueItem(category, value, sourceTag)
	if not (anyScriptExecuted and scanActive) then
		return
	end
	if type(value) ~= "string" then
		return
	end
	local trimmed = value:gsub("^%s+", ""):gsub("%s+$", "")
	if trimmed == "" or #trimmed > 400 then
		return
	end
	if isSelfContent(trimmed) then
		return
	end
	if category == "TEXTS" and isDynamicNoiseText(trimmed) then
		return
	end
	if SEEN[category][trimmed] then
		return
	end
	SEEN[category][trimmed] = true

	table.insert(CAPTURED[category], {
		value = trimmed,
		source = sourceTag or "ScriptExecute",
	})
	notifyUiChanged()
end

local function extractAndCaptureIds(rawStr, sourceTag)
	if not (anyScriptExecuted and scanActive) or type(rawStr) ~= "string" or rawStr == "" then
		return
	end

	for rbxUri in rawStr:gmatch("rbx[a-z]+://[%w%d_%-%./%?&=]+") do
		addUniqueItem("IMAGES", rbxUri, sourceTag)
		addUniqueItem("IDS", rbxUri, sourceTag)
		local idNum = rbxUri:match("(%d+)")
		if idNum and #idNum >= 5 then
			addUniqueItem("IDS", idNum, sourceTag)
		end
	end

	for petId in rawStr:gmatch('[Pp][Ee][Tt][%w_]*%s*[:=]?%s*["\']?(%d%d%d%d%d+)["\']?') do
		addUniqueItem("IDS", petId, sourceTag .. ":Pet")
	end

	if rawStr:match("^%d%d%d%d%d%d+$") and #rawStr <= 18 then
		addUniqueItem("IDS", rawStr, sourceTag)
	end
end

-- =====================================================================
-- 3. ÁP DỤNG LỚP PHỦ THAY THẾ (CHỈ CHẠY KHI ENABLE_CUSTOM_OVERLAY = true)
-- =====================================================================
local function applyStringOverlay(rawStr)
	if not ENABLE_CUSTOM_OVERLAY or type(rawStr) ~= "string" or rawStr == "" then
		return rawStr
	end
	if OVERLAY_CONFIG.TEXT[rawStr] ~= nil then
		return tostring(OVERLAY_CONFIG.TEXT[rawStr])
	end
	if OVERLAY_CONFIG.IDS[rawStr] ~= nil then
		return tostring(OVERLAY_CONFIG.IDS[rawStr])
	end
	if OVERLAY_CONFIG.LINKS[rawStr] ~= nil then
		return tostring(OVERLAY_CONFIG.LINKS[rawStr])
	end
	return rawStr
end

local function applyImageOverlay(rawImg)
	if not ENABLE_CUSTOM_OVERLAY or type(rawImg) ~= "string" or rawImg == "" then
		return rawImg
	end
	if OVERLAY_CONFIG.IMAGES[rawImg] ~= nil then
		return tostring(OVERLAY_CONFIG.IMAGES[rawImg])
	end
	if OVERLAY_CONFIG.IDS[rawImg] ~= nil then
		return tostring(OVERLAY_CONFIG.IDS[rawImg])
	end
	local digits = rawImg:match("(%d+)")
	if digits then
		local rep = OVERLAY_CONFIG.IMAGES[digits] or OVERLAY_CONFIG.IDS[digits]
		if rep ~= nil then
			rep = tostring(rep)
			return rep:match("^rbx") and rep or ("rbxassetid://" .. rep)
		end
	end
	return rawImg
end

local function applyClipboardOverlay(rawClip)
	if not ENABLE_CUSTOM_OVERLAY or type(rawClip) ~= "string" or rawClip == "" then
		return rawClip
	end
	if OVERLAY_CONFIG.CLIPBOARD[rawClip] ~= nil then
		return tostring(OVERLAY_CONFIG.CLIPBOARD[rawClip])
	end
	if OVERLAY_CONFIG.IDS[rawClip] ~= nil then
		return tostring(OVERLAY_CONFIG.IDS[rawClip])
	end
	if OVERLAY_CONFIG.LINKS[rawClip] ~= nil then
		return tostring(OVERLAY_CONFIG.LINKS[rawClip])
	end
	return rawClip
end

-- =====================================================================
-- 4. THEO DÕI SCREENGUI MỚI & QUÉT NỘI DUNG TẢI VỀ

local function snapshotExistingGuis(root)
	if not root then
		return
	end
	pcall(function()
		for _, child in ipairs(root:GetChildren()) do
			existingGuis[child] = true
		end
	end)
end

local function isTrackedTargetDescendant(inst)
	local curr = inst
	while curr do
		if SYSTEM_GUI_BLACKLIST[curr.Name] then
			return false
		end
		if trackedTargetGuis[curr] then
			return true
		end
		curr = curr.Parent
	end
	return false
end

local hookedObjects = setmetatable({}, { __mode = "k" })

local function inspectTargetGuiObject(obj)
	if not obj or not isTrackedTargetDescendant(obj) then
		return
	end

	if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
		local txt = obj.Text
		if type(txt) == "string" and txt ~= "" then
			if scanActive then
				if txt:lower():find("discord%.gg") or txt:lower():find("discord%.com") then
					addUniqueItem("DISCORD", txt, "UI:" .. obj.ClassName)
				elseif txt:match("^https?://") then
					addUniqueItem("URLS", txt, "UI:" .. obj.ClassName)
				else
					addUniqueItem("TEXTS", txt, "UI:" .. obj.ClassName)
				end
				extractAndCaptureIds(txt, "UI:" .. obj.ClassName)
			end
			if ENABLE_CUSTOM_OVERLAY then
				local rep = applyStringOverlay(txt)
				if rep ~= txt then
					obj.Text = rep
				end
			end
		end

		if not hookedObjects[obj] then
			hookedObjects[obj] = true
			obj:GetPropertyChangedSignal("Text"):Connect(function()
				local newTxt = obj.Text
				if type(newTxt) ~= "string" or newTxt == "" then
					return
				end
				if scanActive then
					if newTxt:lower():find("discord%.gg") or newTxt:lower():find("discord%.com") then
						addUniqueItem("DISCORD", newTxt, "UI:" .. obj.ClassName)
					elseif newTxt:match("^https?://") then
						addUniqueItem("URLS", newTxt, "UI:" .. obj.ClassName)
					else
						addUniqueItem("TEXTS", newTxt, "UI:" .. obj.ClassName)
					end
					extractAndCaptureIds(newTxt, "UI:" .. obj.ClassName)
				end
				if ENABLE_CUSTOM_OVERLAY then
					local rep = applyStringOverlay(newTxt)
					if rep ~= newTxt then
						obj.Text = rep
					end
				end
			end)
		end
	elseif obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
		local img = obj.Image
		if type(img) == "string" and img ~= "" then
			if scanActive then
				addUniqueItem("IMAGES", img, "UI:" .. obj.ClassName)
				extractAndCaptureIds(img, "UI:" .. obj.ClassName)
			end
			if ENABLE_CUSTOM_OVERLAY then
				local rep = applyImageOverlay(img)
				if rep ~= img then
					obj.Image = rep
				end
			end
		end

		if not hookedObjects[obj] then
			hookedObjects[obj] = true
			obj:GetPropertyChangedSignal("Image"):Connect(function()
				local newImg = obj.Image
				if type(newImg) ~= "string" or newImg == "" then
					return
				end
				if scanActive then
					addUniqueItem("IMAGES", newImg, "UI:" .. obj.ClassName)
					extractAndCaptureIds(newImg, "UI:" .. obj.ClassName)
				end
				if ENABLE_CUSTOM_OVERLAY then
					local rep = applyImageOverlay(newImg)
					if rep ~= newImg then
						obj.Image = rep
					end
				end
			end)
		end
	end
end

local function registerNewTargetScreenGui(gui)
	if not gui or SYSTEM_GUI_BLACKLIST[gui.Name] or existingGuis[gui] then
		return
	end
	if not gui:IsA("ScreenGui") then
		return
	end
	trackedTargetGuis[gui] = true
	pcall(function()
		for _, desc in ipairs(gui:GetDescendants()) do
			inspectTargetGuiObject(desc)
		end
		gui.DescendantAdded:Connect(function(desc)
			if scanActive or ENABLE_CUSTOM_OVERLAY then
				task.defer(inspectTargetGuiObject, desc)
			end
		end)
	end)
end

local function attachRootContainer(root)
	if not root then
		return
	end
	snapshotExistingGuis(root)
	pcall(function()
		root.ChildAdded:Connect(function(child)
			if anyScriptExecuted then
				task.defer(registerNewTargetScreenGui, child)
			else
				existingGuis[child] = true
			end
		end)
	end)
end

local function startScanSessionForScript(url)
	local h = extractHost(url)
	if h ~= "" then
		activeHosts[h] = true
	end

	-- Nếu chạy một script mới sau khi phiên trước đã dừng, tự mở lại phiên quét cho script mới
	if not anyScriptExecuted or not scanActive then
		anyScriptExecuted = true
		scanActive = true
		scanSessionId = scanSessionId + 1
		local currentSession = scanSessionId
		notifyUiChanged()

		task.delay(AUTO_STOP_SECONDS, function()
			if scanSessionId == currentSession and scanActive then
				scanActive = false
				notifyUiChanged()
			end
		end)
	end
end

local function scanFetchedText(rawStr, sourceTag)
	if not (anyScriptExecuted and scanActive) then
		return
	end
	if type(rawStr) ~= "string" or #rawStr == 0 or isSelfContent(rawStr) then
		return
	end

	local sig = tostring(#rawStr) .. ":" .. rawStr:sub(1, 60) .. ":" .. rawStr:sub(-60)
	if SEEN.SOURCES[sig] then
		return
	end
	SEEN.SOURCES[sig] = true

	for subUrl in rawStr:gmatch("https?://[%w%.%-_~:/%?#@!$&'%(%)%*%+,;=%%]+") do
		local subHost = extractHost(subUrl)
		if subHost ~= "" then
			activeHosts[subHost] = true
		end
		if subUrl:lower():find("discord", 1, true) then
			addUniqueItem("DISCORD", subUrl, sourceTag)
		else
			addUniqueItem("URLS", subUrl, sourceTag)
		end
	end

	for dShort in rawStr:gmatch("discord%.gg/[%w%-_]+") do
		addUniqueItem("DISCORD", dShort, sourceTag)
	end

	extractAndCaptureIds(rawStr, sourceTag)

	for clipStr in rawStr:gmatch('setclipboard%s*%(%s*["\']([^"\'\r\n]+)["\']%s*%)') do
		addUniqueItem("CLIPBOARD", clipStr, sourceTag)
		extractAndCaptureIds(clipStr, sourceTag .. ":Clipboard")
	end
end

if gethui then
	pcall(function()
		attachRootContainer(gethui())
	end)
end
attachRootContainer(CoreGui)
if not PlayerGui and LocalPlayer then
	PlayerGui = LocalPlayer:FindFirstChildOfClass("PlayerGui")
end
if PlayerGui then
	attachRootContainer(PlayerGui)
end

-- =====================================================================
-- 5. HOOK HTTPGET & SETCLIPBOARD (CHẤP NHẬN MỌI SCRIPT ĐƯỢC EXECUTE)
-- =====================================================================
if hookfunction then
	if game.HttpGet then
		pcall(function()
			local oldHttpGet
			oldHttpGet = hookfunction(game.HttpGet, function(self, url, ...)
				local urlStr = type(url) == "string" and url or ""
				if urlStr ~= "" and not isSelfContent(urlStr) then
					startScanSessionForScript(urlStr)
				end

				local res = oldHttpGet(self, url, ...)

				if anyScriptExecuted and scanActive and urlStr ~= "" and not isSelfContent(urlStr) then
					if urlStr:lower():find("discord") then
						addUniqueItem("DISCORD", urlStr, "HttpGet:URL")
					else
						addUniqueItem("URLS", urlStr, "HttpGet:URL")
					end
					extractAndCaptureIds(urlStr, "HttpGet:URL")
				end

				if anyScriptExecuted and scanActive and type(res) == "string" and #res > 0 and not isSelfContent(res) then
					scanFetchedText(res, "HttpGet:" .. urlStr)
				end

				return res
			end)
		end)
	end

	local clipTarget = setclipboard or toclipboard
	if type(clipTarget) == "function" then
		pcall(function()
			local oldClip
			oldClip = hookfunction(clipTarget, function(text, ...)
				if type(text) == "string" and text ~= "" and not isSelfContent(text) then
					if anyScriptExecuted and scanActive then
						if text:lower():find("discord%.gg") or text:lower():find("discord%.com") then
							addUniqueItem("DISCORD", text, "ClipboardButton")
						end
						addUniqueItem("CLIPBOARD", text, "ClipboardButton")
						extractAndCaptureIds(text, "ClipboardButton")
					end
					if ENABLE_CUSTOM_OVERLAY then
						text = applyClipboardOverlay(text)
					end
				end
				return oldClip(text, ...)
			end)
		end)
	end
end

local function escapeLuaStr(s)
	return tostring(s or ""):gsub("\\", "\\\\"):gsub('"', '\\"'):gsub("\r", ""):gsub("\n", "\\n")
end

local function formatEntries(list, mapTable)
	local out = {}
	for _, item in ipairs(list) do
		local chosen = (mapTable and mapTable[item.value] and mapTable[item.value] ~= "") and mapTable[item.value] or item.value
		table.insert(out, string.format('\t\t["%s"] = "%s",', escapeLuaStr(item.value), escapeLuaStr(chosen)))
	end
	return table.concat(out, "\n")
end

	local imageBlock = formatEntries(CAPTURED.IMAGES, OVERLAY_CONFIG.IMAGES)
	local idBlock = formatEntries(CAPTURED.IDS, OVERLAY_CONFIG.IDS)

	for _, item in ipairs(CAPTURED.DISCORD) do
		table.insert(linkEntries, item)
	end
	for _, item in ipairs(CAPTURED.URLS) do
		table.insert(linkEntries, item)
	end
	local linkBlock = formatEntries(linkEntries, OVERLAY_CONFIG.LINKS)
	local clipBlock = formatEntries(CAPTURED.CLIPBOARD, OVERLAY_CONFIG.CLIPBOARD)

		"",
		"local clipFn = setclipboard or toclipboard",
		"if hookfunction and type(clipFn) == \"function\" then",
		"\tpcall(function()",
		"\t\tlocal oldClip",
		"\t\toldClip = hookfunction(clipFn, function(txt, ...)",
		"\t\t\tif type(txt) == \"string\" then txt = mapClipboard(txt) end",
		"\t\t\treturn oldClip(txt, ...)",
		"\t\tend)",
		"\tend)",
		"end",
	}, "\n")
end

local function sanitizeFileName(name)
	local clean = tostring(name or ""):gsub("^%s+", ""):gsub("%s+$", "")
	clean = clean:gsub('[\\/:*?"<>|]', "_")
	return clean
end

local function saveToDeltaFolder(rawName)
	local writeFn = writefile or (syn and syn.write_file) or filewrite
	if not writeFn then
		return false, "Executor không hỗ trợ writefile!"
	end

	local cleanName = sanitizeFileName(rawName)
	if cleanName == "" then
		return false, "Vui lòng đặt tên file trước khi lưu!"
	end

	if not cleanName:lower():match("%.txt$") then
		cleanName = cleanName .. ".txt"
	end

	if type(makefolder) == "function" then
		pcall(function()
			if type(isfolder) == "function" then
				if not isfolder("delta") then
					makefolder("delta")
				end
			else
				makefolder("delta")
			end
		end)
	end

	local fullPath = "delta/" .. cleanName
	local content = buildLightRunnableScript()

	local ok, err = pcall(function()
		writeFn(fullPath, content)
	end)

	if ok then
		return true, "Đã lưu bản nhẹ -> " .. fullPath
	end
	return false, "Lỗi lưu file: " .. tostring(err)
end
