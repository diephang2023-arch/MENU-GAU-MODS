local function isSelfContent(str)
	if type(str) ~= "string" then
		return false
	end
	return string.find(str, SELF_SIGNATURE, 1, true) ~= nil
		or (string.find(str, "GauStudio4WayScannerGui", 1, true) ~= nil and string.find(str, "AUTO_STOP_SECONDS", 1, true) ~= nil)
end

local function totalCapturedCount()
	return #CAPTURED.TEXTS + #CAPTURED.IMAGES + #CAPTURED.IDS + #CAPTURED.DISCORD + #CAPTURED.URLS + #CAPTURED.CLIPBOARD
end

local function notifyUiChanged()
	if type(onStateUpdated) == "function" then
		pcall(onStateUpdated, totalCapturedCount(), scanActive, anyScriptExecuted)
	end
end

local function isDynamicNoiseText(str)
	if #str <= 1 then
		return true
	end
	if str:match("^[%d%s%.,:%-/%%]+$") then
		return true
	end
	local lower = str:lower()
	if lower:match("^fps%s*:") or lower:match("^ping%s*:") or lower:match("%d+%s*ms$") then
		return true
	end
	if lower:match("%d+[%.,]?%d*%s*kg") or lower:match("%d+[%.,]?%d*%s*studs") then
		return true
	end
	if lower:match("^%d+[%.,]?%d*[kmb]$") or lower:match("^%d+[%.,]?%d*x$") then
		return true
	end
	if str:match("^%d+:%d+$") or str:match("^%d+:%d+:%d+$") then
		return true
	end
	return false
end

local function isPetOrItemCardImage(obj)
	if not obj then
		return false
	end
	local nameLower = tostring(obj.Name or ""):lower()
	if nameLower:find("pet") or nameLower:find("item") or nameLower:find("fruit") or nameLower:find("egg") then
		return true
	end
	local curr = obj.Parent
	local depth = 0
	while curr and depth < 4 do
		local pName = tostring(curr.Name or ""):lower()
		if pName:find("pet") or pName:find("inventory") or pName:find("list") or pName:find("card") then
			return true
		end
		if curr:IsA("GuiObject") and not curr:IsA("ScreenGui") then
			for _, sibling in ipairs(curr:GetChildren()) do
				if sibling:IsA("TextLabel") and type(sibling.Text) == "string" then
					local t = sibling.Text:lower()
					if t:match("^%d+[%.,]?%d*[kmb]$")
						or t:match("^%d+[%.,]?%d*x$")
						or t:match("%d+[%.,]?%d*%s*kg")
						or t:find("pet") then
						return true
					end
				end
			end
		end
		curr = curr.Parent
		depth = depth + 1
	end
	return false
end

local function classifyUiTextTag(obj, txt)
	if not obj then
		return ""
	end
	local nameLower = tostring(obj.Name or ""):lower()
	local parentNameLower = obj.Parent and tostring(obj.Parent.Name or ""):lower() or ""
	local txtLower = tostring(txt or ""):lower()

	local curr = obj
	local depth = 0
	while curr and depth < 5 do
		local n = tostring(curr.Name or ""):lower()
		if n:find("splash") or n:find("intro") or n:find("welcome") or n:find("loading") or n:find("boot") then
			return "[MÀN HÌNH CHÀO]"
		end
		curr = curr.Parent
		depth = depth + 1
	end

	if nameLower:find("title") or nameLower:find("header") or nameLower:find("hub") or parentNameLower:find("top") or parentNameLower:find("header") or txtLower:find("hub") then
		return "[TIÊU ĐỀ]"
	end
	if obj:IsA("TextButton") then
		return "[NÚT BẤM]"
	end
	return ""
end

local function classifyUiImageTag(obj)
	if not obj then
		return "[ẢNH UI]"
	end
	local nameLower = tostring(obj.Name or ""):lower()
	local parentNameLower = obj.Parent and tostring(obj.Parent.Name or ""):lower() or ""
	local absSize = Vector2.new(0, 0)
	pcall(function()
		absSize = obj.AbsoluteSize
	end)

	if obj:IsA("ImageButton") then
		if nameLower:find("open")
			or nameLower:find("toggle")
			or nameLower:find("menu")
			or nameLower:find("hub")
			or nameLower:find("mini")
			or nameLower:find("float")
			or parentNameLower:find("open")
			or parentNameLower:find("toggle")
			or parentNameLower:find("mini")
			or (absSize.X > 0 and absSize.X <= 95 and absSize.Y > 0 and absSize.Y <= 95 and obj.Parent and obj.Parent:IsA("ScreenGui")) then
			return "[NÚT MỞ MENU]"
		end
	end

	local curr = obj
	local depth = 0
	while curr and depth < 5 do
		local n = tostring(curr.Name or ""):lower()
		if n:find("splash") or n:find("intro") or n:find("welcome") or n:find("loading") or n:find("boot") or n:find("banner") then
			return "[MÀN HÌNH CHÀO]"
		end
		curr = curr.Parent
		depth = depth + 1
	end

	if absSize.X >= 280 and absSize.Y >= 150 then
		return "[MÀN HÌNH CHÀO]"
	end
	if nameLower:find("logo") or nameLower:find("avatar") or nameLower:find("title") or parentNameLower:find("top") or parentNameLower:find("header") then
		return "[LOGO UI]"
	end
	return "[ẢNH UI]"
end

local function addUniqueItem(category, value, roleTag)
	if not (anyScriptExecuted and scanActive) or type(value) ~= "string" then
		return
	end
	local trimmed = value:gsub("^%s+", ""):gsub("%s+$", "")
	if trimmed == "" or #trimmed > 400 or isSelfContent(trimmed) then
		return
	end
	if category == "TEXTS" and isDynamicNoiseText(trimmed) then
		return
	end

	if category == "TEXTS" then
		local existingIdx = TEXT_INDEX_BY_VAL[trimmed]
		if existingIdx then
			if CAPTURED.TEXTS[existingIdx].tag == "" and roleTag ~= "" then
				CAPTURED.TEXTS[existingIdx].tag = roleTag
			end
			return
		end
		SEEN.TEXTS[trimmed] = true
		table.insert(CAPTURED.TEXTS, { value = trimmed, tag = roleTag or "" })
		TEXT_INDEX_BY_VAL[trimmed] = #CAPTURED.TEXTS
		notifyUiChanged()
		return
	end

	if SEEN[category][trimmed] then
		return
	end
	SEEN[category][trimmed] = true
	table.insert(CAPTURED[category], { value = trimmed, tag = roleTag or "" })
	notifyUiChanged()
end

local function addUiImageWithTag(rawImg, uiTag)
	if not (anyScriptExecuted and scanActive) or type(rawImg) ~= "string" then
		return
	end
	local trimmed = rawImg:gsub("^%s+", ""):gsub("%s+$", "")
	if trimmed == "" or #trimmed > 300 or isSelfContent(trimmed) then
		return
	end

	local existingIdx = IMAGE_INDEX_BY_VAL[trimmed]
	if existingIdx then
		if CAPTURED.IMAGES[existingIdx].tag == "[ẢNH UI]" and uiTag ~= "[ẢNH UI]" then
			CAPTURED.IMAGES[existingIdx].tag = uiTag
		end
		return
	end

	SEEN.IMAGES[trimmed] = true
	table.insert(CAPTURED.IMAGES, { value = trimmed, tag = uiTag or "[ẢNH UI]" })
	IMAGE_INDEX_BY_VAL[trimmed] = #CAPTURED.IMAGES

	local idNum = trimmed:match("(%d+)")
	if idNum and #idNum >= 5 and not SEEN.IDS[idNum] then
		SEEN.IDS[idNum] = true
		table.insert(CAPTURED.IDS, { value = idNum, tag = uiTag or "[ẢNH UI]" })
	end
	notifyUiChanged()
end

local function applyStringOverlay(rawStr)
	if not ENABLE_CUSTOM_OVERLAY or type(rawStr) ~= "string" or rawStr == "" then
		return rawStr
	end
	return OVERLAY_CONFIG.TEXT[rawStr] or OVERLAY_CONFIG.IDS[rawStr] or OVERLAY_CONFIG.LINKS[rawStr] or rawStr
end

local function applyImageOverlay(rawImg)
	if not ENABLE_CUSTOM_OVERLAY or type(rawImg) ~= "string" or rawImg == "" then
		return rawImg
	end
	local direct = OVERLAY_CONFIG.IMAGES[rawImg] or OVERLAY_CONFIG.IDS[rawImg]
	if direct ~= nil then
		return tostring(direct)
	end
	local digits = rawImg:match("(%d+)")
	if digits then
		local rep = OVERLAY_CONFIG.IMAGES[digits] or OVERLAY_CONFIG.IDS[digits]
		if rep ~= nil then
			rep = tostring(rep)
			return rep:match("^rbx") and rep or ("rbxassetid://" .. rep)
		end
	end
	return rawImg
end

local function applyClipboardOverlay(rawClip)
	if not ENABLE_CUSTOM_OVERLAY or type(rawClip) ~= "string" or rawClip == "" then
		return rawClip
	end
	return OVERLAY_CONFIG.CLIPBOARD[rawClip] or OVERLAY_CONFIG.IDS[rawClip] or OVERLAY_CONFIG.LINKS[rawClip] or rawClip
end

local function snapshotExistingGuis(root)
	if not root then return end
	pcall(function()
		for _, child in ipairs(root:GetChildren()) do
			existingGuis[child] = true
		end
	end)
end

local function isTrackedTargetDescendant(inst)
	local curr = inst
	while curr do
		if SYSTEM_GUI_BLACKLIST[curr.Name] then return false end
		if trackedTargetGuis[curr] then return true end
		curr = curr.Parent
	end
	return false
end

local hookedObjects = setmetatable({}, { __mode = "k" })
local trackedGuiObjects = setmetatable({}, { __mode = "k" })

local function inspectTargetGuiObject(obj)
	if not obj or not isTrackedTargetDescendant(obj) then
		return
	end

	if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
		trackedGuiObjects[obj] = true
		local txt = obj.Text
		if type(txt) == "string" and txt ~= "" then
			if scanActive then
				if txt:lower():find("discord%.gg") or txt:lower():find("discord%.com") then
					addUniqueItem("DISCORD", txt, "[LINK DISCORD]")
				elseif txt:match("^https?://") then
					addUniqueItem("URLS", txt, "[LINK]")
				else
					addUniqueItem("TEXTS", txt, classifyUiTextTag(obj, txt))
				end
			end
			if ENABLE_CUSTOM_OVERLAY then
				local rep = applyStringOverlay(txt)
				if rep ~= txt then obj.Text = rep end
			end
		end

		if not hookedObjects[obj] then
			hookedObjects[obj] = true
			obj:GetPropertyChangedSignal("Text"):Connect(function()
				local newTxt = obj.Text
				if type(newTxt) ~= "string" or newTxt == "" then return end
				if scanActive then
					if newTxt:lower():find("discord%.gg") or newTxt:lower():find("discord%.com") then
						addUniqueItem("DISCORD", newTxt, "[LINK DISCORD]")
					elseif newTxt:match("^https?://") then
						addUniqueItem("URLS", newTxt, "[LINK]")
					else
						addUniqueItem("TEXTS", newTxt, classifyUiTextTag(obj, newTxt))
					end
				end
				if ENABLE_CUSTOM_OVERLAY then
					local rep = applyStringOverlay(newTxt)
					if rep ~= newTxt then obj.Text = rep end
				end
			end)
		end
	elseif obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
		if isPetOrItemCardImage(obj) then return end
		trackedGuiObjects[obj] = true

		local img = obj.Image
		if type(img) == "string" and img ~= "" then
			if scanActive then
				addUiImageWithTag(img, classifyUiImageTag(obj))
			end
			if ENABLE_CUSTOM_OVERLAY then
				local rep = applyImageOverlay(img)
				if rep ~= img then obj.Image = rep end
			end
		end

		if not hookedObjects[obj] then
			hookedObjects[obj] = true
			obj:GetPropertyChangedSignal("Image"):Connect(function()
				if isPetOrItemCardImage(obj) then return end
				local newImg = obj.Image
				if type(newImg) == "string" and newImg ~= "" then
					if scanActive then
						addUiImageWithTag(newImg, classifyUiImageTag(obj))
					end
					if ENABLE_CUSTOM_OVERLAY then
						local rep = applyImageOverlay(newImg)
						if rep ~= newImg then obj.Image = rep end
					end
				end
			end)
		end
	end
end

livePatchAllRoots = function()
	if not ENABLE_CUSTOM_OVERLAY then return end
	pcall(function()
		for obj in pairs(trackedGuiObjects) do
			if obj and obj.Parent then
				if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
					local rep = applyStringOverlay(obj.Text)
					if rep ~= obj.Text then obj.Text = rep end
				elseif (obj:IsA("ImageLabel") or obj:IsA("ImageButton")) and not isPetOrItemCardImage(obj) then
					local rep = applyImageOverlay(obj.Image)
					if rep ~= obj.Image then obj.Image = rep end
				end
			end
		end
	end)
end

local function registerNewTargetScreenGui(gui)
	if not gui or SYSTEM_GUI_BLACKLIST[gui.Name] or existingGuis[gui] or not gui:IsA("ScreenGui") then
		return
	end
	trackedTargetGuis[gui] = true
	pcall(function()
		for _, desc in ipairs(gui:GetDescendants()) do
			inspectTargetGuiObject(desc)
		end
		gui.DescendantAdded:Connect(function(desc)
			if scanActive or ENABLE_CUSTOM_OVERLAY then
				task.defer(inspectTargetGuiObject, desc)
			end
		end)
	end)
end

local function attachRootContainer(root)
	if not root then return end
	snapshotExistingGuis(root)
	pcall(function()
		root.ChildAdded:Connect(function(child)
			if anyScriptExecuted then
				task.defer(registerNewTargetScreenGui, child)
			else
				existingGuis[child] = true
			end
		end)
	end)
end

local function startScanSessionForScript()
	if anyScriptExecuted and scanActive then return end
	anyScriptExecuted = true
	scanActive = true
	scanSessionId = scanSessionId + 1
	local currentSession = scanSessionId
	notifyUiChanged()

	task.delay(AUTO_STOP_SECONDS, function()
		if scanSessionId == currentSession and scanActive then
			scanActive = false
			notifyUiChanged()
		end
	end)
end

local function scanFetchedText(rawStr)
	if not (anyScriptExecuted and scanActive) or type(rawStr) ~= "string" or #rawStr == 0 or isSelfContent(rawStr) then
		return
	end
	local sig = tostring(#rawStr) .. ":" .. rawStr:sub(1, 60) .. ":" .. rawStr:sub(-60)
	if SEEN.SOURCES[sig] then return end
	SEEN.SOURCES[sig] = true

	for subUrl in rawStr:gmatch("https?://[%w%.%-_~:/%?#@!$&'%(%)%*%+,;=%%]+") do
		if subUrl:lower():find("discord", 1, true) then
			addUniqueItem("DISCORD", subUrl, "[LINK DISCORD]")
		else
			addUniqueItem("URLS", subUrl, "[LINK]")
		end
	end
	for dShort in rawStr:gmatch("discord%.gg/[%w%-_]+") do
		addUniqueItem("DISCORD", dShort, "[LINK DISCORD]")
	end
	for clipStr in rawStr:gmatch('setclipboard%s*%(%s*["\']([^"\'\r\n]+)["\']%s*%)') do
		addUniqueItem("CLIPBOARD", clipStr, "[NÚT COPY]")
	end
end

if hookfunction then
	if game.HttpGet then
		pcall(function()
			local oldHttpGet
			oldHttpGet = hookfunction(game.HttpGet, function(self, url, ...)
				local urlStr = type(url) == "string" and url or ""
				if urlStr ~= "" and not isSelfContent(urlStr) then
					startScanSessionForScript()
				end
				local res = oldHttpGet(self, url, ...)
				if anyScriptExecuted and scanActive and urlStr ~= "" and not isSelfContent(urlStr) then
					if urlStr:lower():find("discord") then
						addUniqueItem("DISCORD", urlStr, "[LINK DISCORD]")
					else
						addUniqueItem("URLS", urlStr, "[LINK]")
					end
				end
				if anyScriptExecuted and scanActive and type(res) == "string" and #res > 0 and not isSelfContent(res) then
					scanFetchedText(res)
				end
				return res
			end)
		end)
	end

	local clipTarget = setclipboard or toclipboard
	if type(clipTarget) == "function" then
		pcall(function()
			local oldClip
			oldClip = hookfunction(clipTarget, function(text, ...)
				if type(text) == "string" and text ~= "" and not isSelfContent(text) then
					if anyScriptExecuted and scanActive then
						if text:lower():find("discord%.gg") or text:lower():find("discord%.com") then
							addUniqueItem("DISCORD", text, "[LINK DISCORD]")
						end
						addUniqueItem("CLIPBOARD", text, "[NÚT COPY]")
					end
					if ENABLE_CUSTOM_OVERLAY then
						text = applyClipboardOverlay(text)
					end
				end
				return oldClip(text, ...)
			end)
		end)
	end
end

local function escapeLuaStr(s)
	return tostring(s or ""):gsub("\\", "\\\\"):gsub('"', '\\"'):gsub("\r", ""):gsub("\n", "\\n")
end

local function formatEasyEntries(list, mapTable)
	local out = {}
	for _, item in ipairs(list) do
		local origVal = tostring(item.value or "")
		local chosen = (mapTable and mapTable[origVal] and mapTable[origVal] ~= "") and tostring(mapTable[origVal]) or origVal
		local tag = (item.tag and item.tag ~= "") and (" -- " .. tostring(item.tag)) or ""
		table.insert(out, string.format('\t\t["%s"] = "%s",%s', escapeLuaStr(origVal), escapeLuaStr(chosen), tag))
	end
	return table.concat(out, "\n")
end

local function buildEasyRunnableScript(sectionFilter)
	local filter = sectionFilter or "ALL"
	local textBlock = formatEasyEntries(CAPTURED.TEXTS, OVERLAY_CONFIG.TEXT)
	local imageBlock = formatEasyEntries(CAPTURED.IMAGES, OVERLAY_CONFIG.IMAGES)
	local idBlock = formatEasyEntries(CAPTURED.IDS, OVERLAY_CONFIG.IDS)
	local linkEntries = {}
	for _, item in ipairs(CAPTURED.DISCORD) do table.insert(linkEntries, item) end
	for _, item in ipairs(CAPTURED.URLS) do table.insert(linkEntries, item) end
	local linkBlock = formatEasyEntries(linkEntries, OVERLAY_CONFIG.LINKS)
	local clipBlock = formatEasyEntries(CAPTURED.CLIPBOARD, OVERLAY_CONFIG.CLIPBOARD)

	local function blockOrPlaceholder(name, block, fallback)
		if filter ~= "ALL" and filter ~= name then
			return "\t\t-- (Đang ẩn mục " .. name .. " ở tab lọc hiện tại)"
		end
		return block ~= "" and block or fallback
	end

	return table.concat({
		"--[[ EASY OVERLAY SCRIPT | GAU MODS ]]",
		"-- Chỉ cần sửa chữ/số trong ngoặc kép THỨ HAI (sau dấu =), rồi execute!",
		"",
		"local Players = game:GetService(\"Players\")",
		"local CoreGui = game:GetService(\"CoreGui\")",
		"local LocalPlayer = Players.LocalPlayer",
		"local PlayerGui = LocalPlayer and LocalPlayer:FindFirstChildOfClass(\"PlayerGui\")",
		"",
		"local ENABLE_CUSTOM_OVERLAY = true",
		"",
		"local OVERLAY_CONFIG = {",
		"\tTEXT = {",
		blockOrPlaceholder("TEXT", textBlock, '\t\t-- ["Chữ cũ"] = "Chữ mới",'),
		"\t},",
		"\tIMAGES = {",
		blockOrPlaceholder("IMAGES", imageBlock, '\t\t-- ["rbxassetid://123456"] = "rbxassetid://654321", -- [NÚT MỞ MENU]'),
		"\t},",
		"\tIDS = {",
		blockOrPlaceholder("IDS", idBlock, '\t\t-- ["123456"] = "654321", -- [NÚT MỞ MENU]'),
		"\t},",
		"\tLINKS = {",

-- Đọc lại các cặp ["cũ"] = "mới" người dùng đã sửa trong khung code để cập nhật OVERLAY_CONFIG trước khi lưu/áp dụng
local function syncOverlayConfigFromEditedCode(codeText)
	if type(codeText) ~= "string" or codeText == "" then
		return
	end
	local currentSection = nil
	for rawLine in (codeText .. "\n"):gmatch("([^\r\n]*)\r?\n") do
		local line = rawLine:gsub("^%s+", "")
		if line:match("^TEXT%s*=%s*{") then
			currentSection = "TEXT"
		elseif line:match("^IMAGES%s*=%s*{") then
			currentSection = "IMAGES"
		elseif line:match("^IDS%s*=%s*{") then
			currentSection = "IDS"
		elseif line:match("^LINKS%s*=%s*{") then
			currentSection = "LINKS"
		elseif line:match("^CLIPBOARD%s*=%s*{") then
			currentSection = "CLIPBOARD"
		elseif line:match("^}%s*,?") then
			currentSection = nil
		elseif currentSection and not line:match("^%-%-") then
			local k, v = line:match('^%[%"(.-)%"%]%s*=%s*%"(.-)%"')
			if k and v then
				OVERLAY_CONFIG[currentSection][k] = v
			end
		end
	end
end

local function sanitizeFileName(name)
	local clean = tostring(name or ""):gsub("^%s+", ""):gsub("%s+$", "")
	clean = clean:gsub('[\\/:*?"<>|]', "_")
	return clean
end

local function saveCustomCodeToDeltaFolder(rawName, codeContent)
	local writeFn = writefile or (syn and syn.write_file) or filewrite
	if not writeFn then
		return false, "Executor không hỗ trợ writefile!"
	end

	local cleanName = sanitizeFileName(rawName)
	if cleanName == "" then
		return false, "Vui lòng nhập tên file trước khi lưu!"
	end

	if not cleanName:lower():match("%.txt$") then
		cleanName = cleanName .. ".txt"
	end

	if type(makefolder) == "function" then
		pcall(function()
			if type(isfolder) == "function" then
				if not isfolder("delta") then makefolder("delta") end
			else
				makefolder("delta")
			end
		end)
	end

	local fullPath = "delta/" .. cleanName
	local finalContent = (type(codeContent) == "string" and codeContent ~= "") and codeContent or buildEasyRunnableScript("ALL")

	local ok, err = pcall(function()
		writeFn(fullPath, finalContent)
	end)

	if ok then
		return true, "Đã lưu thành công -> " .. fullPath
	end
	return false, "Lỗi lưu file: " .. tostring(err)
end

		local widthPx = math.max(960, maxLen * 8 + 80)
		local heightPx = math.max(600, lineCount * 17 + 80)
		codeBox.Size = UDim2.fromOffset(widthPx, heightPx)
		codeScroll.CanvasSize = UDim2.fromOffset(widthPx + 24, heightPx + 24)
	end

	local searchMatches = {}
	local activeMatchIdx = 0

	local function jumpToMatch(index)
		if #searchMatches == 0 then
			matchCountLbl.Text = "0/0"
			return
		end
		if index < 1 then
			index = #searchMatches
		elseif index > #searchMatches then
			index = 1
		end
		activeMatchIdx = index
		local hit = searchMatches[activeMatchIdx]
		matchCountLbl.Text = tostring(activeMatchIdx) .. "/" .. tostring(#searchMatches)

		local targetY = math.max(0, (hit.line - 2) * 17)
		local targetX = math.max(0, (hit.col - 1) * 7)
		local maxY = math.max(0, codeScroll.CanvasSize.Y.Offset - codeScroll.AbsoluteSize.Y)
		local maxX = math.max(0, codeScroll.CanvasSize.X.Offset - codeScroll.AbsoluteSize.X)
		codeScroll.CanvasPosition = Vector2.new(math.clamp(targetX, 0, maxX), math.clamp(targetY, 0, maxY))
	end

	local function rebuildSearchResults()
		searchMatches = {}
		activeMatchIdx = 0

		for _, child in ipairs(hitBar:GetChildren()) do
			if child:IsA("TextButton") then
				child:Destroy()
			end
		end

		local query = tostring(searchBox.Text or ""):lower()
		query = query:gsub("^%s+", ""):gsub("%s+$", "")
		if query == "" then
			matchCountLbl.Text = "0/0"
			hitBar.CanvasSize = UDim2.fromOffset(100, 24)
			return
		end

		local lineNum = 0
		local totalWidth = 0
		for line in (codeBox.Text .. "\n"):gmatch("([^\r\n]*)\r?\n") do
			lineNum = lineNum + 1
			local sPos = line:lower():find(query, 1, true)
			if sPos then
				table.insert(searchMatches, { line = lineNum, col = sPos, text = line })
				local mIdx = #searchMatches
				local chip = Instance.new("TextButton")
				local preview = line:gsub("^%s+", "")
				if #preview > 26 then
					preview = preview:sub(1, 24) .. ".."
				end
				chip.Size = UDim2.fromOffset(156, 20)
				chip.Position = UDim2.fromOffset(0, 2)
				chip.BackgroundColor3 = Color3.fromRGB(44, 56, 80)
				chip.TextColor3 = Color3.fromRGB(255, 232, 130)
				chip.Font = Enum.Font.Code
				chip.TextSize = 10
				chip.Text = "D" .. tostring(lineNum) .. ": " .. preview
				chip.BorderSizePixel = 0
				chip.Parent = hitBar
				Instance.new("UICorner", chip).CornerRadius = UDim.new(0, 5)
				totalWidth = totalWidth + 161

				chip.MouseButton1Click:Connect(function()
					jumpToMatch(mIdx)
				end)
			end
		end

		hitBar.CanvasSize = UDim2.fromOffset(math.max(120, totalWidth + 10), 24)
		if #searchMatches > 0 then
			jumpToMatch(1)
		else
			matchCountLbl.Text = "0/0"
		end
	end

	local function refreshTabStyles()
		for key, btn in pairs(tabButtons) do
			if key == currentFilter then
				btn.BackgroundColor3 = Color3.fromRGB(85, 140, 255)
				btn.TextColor3 = Color3.fromRGB(255, 255, 255)
			else
				btn.BackgroundColor3 = Color3.fromRGB(34, 39, 52)
				btn.TextColor3 = Color3.fromRGB(175, 184, 205)
			end
		end
	end

	local function loadScriptIntoEditor(filterKey)
		currentFilter = filterKey or "ALL"
		refreshTabStyles()
		codeBox.Text = buildEasyRunnableScript(currentFilter)
		resizeCodeCanvas()
		rebuildSearchResults()
	end

	for _, tab in ipairs(filterTabs) do
		local tBtn = Instance.new("TextButton")
		tBtn.Size = UDim2.fromOffset(68, 24)
		tBtn.BackgroundColor3 = Color3.fromRGB(34, 39, 52)
		tBtn.TextColor3 = Color3.fromRGB(175, 184, 205)
		tBtn.Font = Enum.Font.GothamBold
		tBtn.TextSize = 10
		tBtn.Text = tab.label
		tBtn.BorderSizePixel = 0
		tBtn.Parent = filterRow
		Instance.new("UICorner", tBtn).CornerRadius = UDim.new(0, 6)
		tabButtons[tab.key] = tBtn

		tBtn.MouseButton1Click:Connect(function()
			syncOverlayConfigFromEditedCode(codeBox.Text)
			loadScriptIntoEditor(tab.key)
		end)
	end

	searchBox:GetPropertyChangedSignal("Text"):Connect(rebuildSearchResults)
	codeBox:GetPropertyChangedSignal("Text"):Connect(resizeCodeCanvas)

	prevMatchBtn.MouseButton1Click:Connect(function()
		jumpToMatch(activeMatchIdx - 1)
	end)

	nextMatchBtn.MouseButton1Click:Connect(function()
		jumpToMatch(activeMatchIdx + 1)
	end)

	clearSearchBtn.MouseButton1Click:Connect(function()
		searchBox.Text = ""
		rebuildSearchResults()
	end)

	applyLiveBtn.MouseButton1Click:Connect(function()
		syncOverlayConfigFromEditedCode(codeBox.Text)
		if livePatchAllRoots then
			livePatchAllRoots()
		end
		statusBanner.TextColor3 = Color3.fromRGB(115, 215, 255)
		statusBanner.Text = "Đã đồng bộ & áp dụng cấu hình trực tiếp lên UI!"
	end)

	copyBtn.MouseButton1Click:Connect(function()
		local clipFn = setclipboard or toclipboard
		if type(clipFn) == "function" then
			pcall(clipFn, codeBox.Text)
			statusBanner.TextColor3 = Color3.fromRGB(120, 235, 150)
			statusBanner.Text = "Đã sao chép toàn bộ mã Lua vào clipboard!"
		else
			statusBanner.TextColor3 = Color3.fromRGB(255, 125, 125)
			statusBanner.Text = "Executor không hỗ trợ clipboard!"
		end
	end)

	closeBtn.MouseButton1Click:Connect(function()
		studio.Visible = false
	end)

	saveConfirmBtn.MouseButton1Click:Connect(function()
		syncOverlayConfigFromEditedCode(codeBox.Text)
		local ok, msg = saveCustomCodeToDeltaFolder(fileNameBox.Text, codeBox.Text)
		if ok then
			statusBanner.TextColor3 = Color3.fromRGB(105, 240, 145)
			statusBanner.Text = msg
			task.delay(0.9, function()
				if studio then studio.Visible = false end
			end)
		else
			statusBanner.TextColor3 = Color3.fromRGB(255, 115, 115)
			statusBanner.Text = msg
			fileNameBox:CaptureFocus()
		end
	end)

	-- Kéo thanh nút nổi
	local barDragging, barDragStart, barStartPos, barMoved = false, nil, nil, false
	floatBtn.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
			barDragging = true
			barMoved = false
			barDragStart = input.Position
			barStartPos = holder.Position
		end
	end)

	-- Kéo 4 chiều trực tiếp trên khung codeScroll (Lên / Xuống / Trái / Phải)
	local panDragging = false
	local panStartInputPos = nil
	local panStartCanvasPos = nil
	local panMoved = false

	codeScroll.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
			panDragging = true
			panMoved = false
			panStartInputPos = input.Position
			panStartCanvasPos = codeScroll.CanvasPosition
		end
	end)

	-- Kéo di chuyển cả cửa sổ Studio từ thanh TopBar
	local winDragging = false
	local winDragStart = nil
	local winStartPos = nil

	topBar.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
			winDragging = true
			winDragStart = input.Position
			winStartPos = studio.Position
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement then
			if barDragging then
				local delta = input.Position - barDragStart
				if delta.Magnitude > 8 then barMoved = true end
				holder.Position = UDim2.new(
					barStartPos.X.Scale,
					barStartPos.X.Offset + delta.X,
					barStartPos.Y.Scale,
					barStartPos.Y.Offset + delta.Y
				)
			elseif panDragging then
				local delta = input.Position - panStartInputPos
				if delta.Magnitude > 6 then
					panMoved = true
				end
				local maxX = math.max(0, codeScroll.CanvasSize.X.Offset - codeScroll.AbsoluteSize.X)
				local maxY = math.max(0, codeScroll.CanvasSize.Y.Offset - codeScroll.AbsoluteSize.Y)
				local nextX = math.clamp(panStartCanvasPos.X - delta.X, 0, maxX)
				local nextY = math.clamp(panStartCanvasPos.Y - delta.Y, 0, maxY)
				codeScroll.CanvasPosition = Vector2.new(nextX, nextY)
			elseif winDragging then
				local delta = input.Position - winDragStart
				studio.Position = UDim2.new(
					winStartPos.X.Scale,
					winStartPos.X.Offset + delta.X,
					winStartPos.Y.Scale,
					winStartPos.Y.Offset + delta.Y
				)
			end
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
			if barDragging then
				barDragging = false
				if not barMoved then
					if not studio.Visible then
						statusBanner.TextColor3 = Color3.fromRGB(150, 210, 175)
						statusBanner.Text = "Kéo 4 chiều tự do • Sửa trực tiếp rồi bấm Lưu hoặc Áp dụng"
						loadScriptIntoEditor("ALL")
						studio.Visible = true
					else
						studio.Visible = false
					end
				end
			end
			if panDragging then
				panDragging = false
				if not panMoved then
					codeBox:CaptureFocus()
				end
			end
			if winDragging then
				winDragging = false
			end
		end
	end)
end
