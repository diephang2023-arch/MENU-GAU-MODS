local UUID_RE = "(%x%x%x%x%x%x%x%x%-%x%x%x%x%-%x%x%x%x%-%x%x%x%x%-%x%x%x%x%x%x%x%x%x%x%x%x)"
local function extractCFJobIds(body)
    if type(body) ~= "string" then return end
    local ids = {}
    for uuid in body:gmatch(UUID_RE) do
        if uuid ~= game.JobId then
            local dup = false
            for _, existing in ipairs(ids) do if existing == uuid then dup=true; break end end
            if not dup then table.insert(ids, uuid) end
        end
    end
    if #ids >= 1 then
        _G.__CF_JobIds      = ids
        _G.__CF_JobIds_tick = tick()
    end
end

local lockedTogglePanel = nil

local function isProtectedPanel(obj)
    if not obj then return true end
    if obj.Name == "GauMods_LoadError" then return true end
    local curr = obj
    while curr do
        if curr:IsA("GuiObject") then
            local n = (curr.Name or ""):lower()
            if n:find("steal") or n:find("panel") or n:find("finder") or n:find("chilli")
            or n:find("hud") or n:find("quick") or n:find("cash") or n:find("money")
            or n:find("shop") or n:find("index") or n:find("prompt") then
                return true
            end
        end
        if curr:IsA("ScreenGui") then
            local sn = (curr.Name or ""):lower()
            if sn:find("steal") or sn:find("panel") or sn:find("finder") or sn:find("chilli")
            or sn:find("hud") or sn:find("quick") or sn:find("prompt") then
                return true
            end
            break
        end
        curr = curr.Parent
    end
    return false
end

local function cleanupStrayLogos()
    for _, root in ipairs(roots) do
        if root then
            for _, desc in ipairs(root:GetDescendants()) do
                if desc.Name == "GauCoverLogo" or desc.Name == "ChilliV5CoverLogo" or desc.Name == "InnerChilliLogo" then
                    if not lockedTogglePanel or not desc:IsDescendantOf(lockedTogglePanel) then
                        pcall(function() desc:Destroy() end)
                    end
                elseif desc.Name == "RainbowBorder" then
                    if not lockedTogglePanel or not desc:IsDescendantOf(lockedTogglePanel) then
                        pcall(function() desc:Destroy() end)
                    end
                end
            end
        end
    end
end

local function clampOverlayBounds(overlay, parentObj)
    if not overlay or not overlay.Parent then return end
    overlay.AnchorPoint = Vector2.new(0.5, 0.5)
    overlay.Position = UDim2.new(0.5, 0, 0.5, 0)
    overlay.Size = UDim2.new(1, 0, 1, 0)
    if not overlay:FindFirstChild("GauMaxSize") then
        local maxLimit = Instance.new("UISizeConstraint")
        maxLimit.Name = "GauMaxSize"
        maxLimit.MaxSize = Vector2.new(TOGGLE_SIZE_PX, TOGGLE_SIZE_PX)
        maxLimit.MinSize = Vector2.new(24, 24)
        maxLimit.Parent = overlay
    end
    if parentObj and parentObj:IsA("GuiObject") then
        local sz = parentObj.AbsoluteSize
        if sz.X > TOGGLE_SIZE_PX + 10 or sz.Y > TOGGLE_SIZE_PX + 10 then
            parentObj.Size = UDim2.fromOffset(TOGGLE_SIZE_PX, TOGGLE_SIZE_PX)
        end
    end
end

local function setupCleanToggleLogo(targetObj)
    if not (targetObj and targetObj:IsA("GuiObject")) then return end
    if isProtectedPanel(targetObj) then return end
    if targetObj:FindFirstAncestorOfClass("ScrollingFrame") then return end
    if targetObj:FindFirstAncestorOfClass("ProximityPrompt") then return end

    -- Nếu đã khóa 1 nút mở menu chính rồi thì chỉ giữ/cập nhật đúng nút đó
    if lockedTogglePanel and lockedTogglePanel.Parent then
        if targetObj ~= lockedTogglePanel and not targetObj:IsDescendantOf(lockedTogglePanel) then
            return
        end
    else
        lockedTogglePanel = nil
    end

    local topPanel = targetObj
    while topPanel.Parent and topPanel.Parent:IsA("GuiObject") and not topPanel.Parent:IsA("ScreenGui") do
        local pSz = topPanel.Parent.AbsoluteSize
        if pSz.X > 120 or pSz.Y > 120 then break end
        topPanel = topPanel.Parent
    end

    if isProtectedPanel(topPanel) then return end

    local sz = topPanel.AbsoluteSize
    local pos = topPanel.AbsolutePosition
    if sz.X <= 0 or sz.Y <= 0 then return end
    -- Chỉ nhận nút toggle nằm ở vùng trên/trái màn hình với kích thước gọn vuông vức
    if sz.X > 95 or sz.Y > 95 or sz.X < 34 or sz.Y < 34 then return end
    if math.abs(sz.X - sz.Y) > 18 or pos.X > 220 or pos.Y > 260 then return end

    lockedTogglePanel = topPanel
    cleanupStrayLogos()

    if not topPanel:GetAttribute("__GauToggleFixed") then
        topPanel:SetAttribute("__GauToggleFixed", true)

        for _, child in ipairs(topPanel:GetDescendants()) do
            if child:IsA("UIPadding") or child:IsA("UIAspectRatioConstraint") or child:IsA("UISizeConstraint") then
                if child.Name ~= "PerfectCircleAspect" and child.Name ~= "GauMaxSize" and child.Name ~= "GauPanelMax" then
                    pcall(function() child:Destroy() end)
                end
            end
            if child.Name ~= "GauCoverLogo" then
                if child:IsA("ImageLabel") or child:IsA("ImageButton") then
                    child.ImageTransparency = 1
                    child.BackgroundTransparency = 1
                elseif child:IsA("TextLabel") or child:IsA("TextButton") then
                    child:SetAttribute("__HiddenVxLogo", true)
                    child.TextTransparency = 1
                    child.BackgroundTransparency = 1
                end
            end
        end

        if topPanel:IsA("ImageLabel") or topPanel:IsA("ImageButton") then
            topPanel.ImageTransparency = 1
        elseif topPanel:IsA("TextLabel") or topPanel:IsA("TextButton") then
            topPanel:SetAttribute("__HiddenVxLogo", true)
            topPanel.TextTransparency = 1
        end

        topPanel.Size = UDim2.fromOffset(TOGGLE_SIZE_PX, TOGGLE_SIZE_PX)
        topPanel.BackgroundColor3 = LOGO_BG_COLOR
        topPanel.BackgroundTransparency = 0
        topPanel.BorderSizePixel = 0
        topPanel.ClipsDescendants = true

        local aspect = topPanel:FindFirstChild("PerfectCircleAspect") or Instance.new("UIAspectRatioConstraint")
        aspect.Name = "PerfectCircleAspect"
        aspect.AspectRatio = 1
        aspect.Parent = topPanel

        local panelMax = topPanel:FindFirstChild("GauPanelMax") or Instance.new("UISizeConstraint")
        panelMax.Name = "GauPanelMax"
        panelMax.MaxSize = Vector2.new(TOGGLE_SIZE_PX, TOGGLE_SIZE_PX)
        panelMax.MinSize = Vector2.new(34, 34)
        panelMax.Parent = topPanel

        local cCorner = topPanel:FindFirstChildOfClass("UICorner") or Instance.new("UICorner")
        cCorner.CornerRadius = UDim.new(1, 0)
        cCorner.Parent = topPanel

        topPanel:GetPropertyChangedSignal("Size"):Connect(function()
            if topPanel.Size ~= UDim2.fromOffset(TOGGLE_SIZE_PX, TOGGLE_SIZE_PX) then
                topPanel.Size = UDim2.fromOffset(TOGGLE_SIZE_PX, TOGGLE_SIZE_PX)

        local stroke = topPanel:FindFirstChild("RainbowBorder") or Instance.new("UIStroke")

        stroke.Thickness = 2
        stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border

    if not coverLogo then
        coverLogo = Instance.new("ImageLabel")
        coverLogo.Name = "GauCoverLogo"
        coverLogo.Parent = topPanel
    end

    coverLogo.BackgroundTransparency = 1
    coverLogo.Image = CHILLI_LOGO_ID
    coverLogo.ImageTransparency = 0
    coverLogo.ImageColor3 = Color3.fromRGB(255, 255, 255)
    coverLogo.ScaleType = Enum.ScaleType.Fit

local function patchHeaderLogoLeftOfTitle(headerContainer, titleLbl)
    if not headerContainer or not titleLbl then return end
    if isProtectedPanel(headerContainer) then return end

    local titleX = titleLbl.AbsolutePosition.X
    for _, item in ipairs(headerContainer:GetDescendants()) do
        if item ~= titleLbl and item:IsA("GuiObject") and item.Name ~= "GauHeaderLogo" then
            local pos = item.AbsolutePosition
            local sz = item.AbsoluteSize
            if math.abs(pos.Y - headerY) < 35 and pos.X < titleX and sz.X >= 10 and sz.X <= 42 and sz.Y <= 42 then

                    item.Image = CHILLI_LOGO_ID
                    item.ImageTransparency = 0
                    item.ImageColor3 = Color3.fromRGB(255, 255, 255)
                    item.BackgroundTransparency = 1
                    item.ScaleType = Enum.ScaleType.Fit
                elseif item:IsA("TextLabel") or item:IsA("TextButton") then
                    item:SetAttribute("__HiddenVxLogo", true)
                    item.TextTransparency = 1
                    local holder = item.Parent
                    if holder and not holder:FindFirstChild("GauHeaderLogo") then
                        local hLogo = Instance.new("ImageLabel")
                        hLogo.Name = "GauHeaderLogo"
                        hLogo.Size = UDim2.fromOffset(22, 22)
                        hLogo.Position = UDim2.new(0.5, 0, 0.5, 0)
                        hLogo.AnchorPoint = Vector2.new(0.5, 0.5)
                        hLogo.BackgroundTransparency = 1
                        hLogo.Image = CHILLI_LOGO_ID
                        hLogo.ScaleType = Enum.ScaleType.Fit
                        hLogo.ZIndex = math.min((item.ZIndex or 1) + 2, 15)
                        hLogo.Parent = holder
                    end
                end
            end
        end
    end
end

    if not (obj:IsA("TextLabel") or obj:IsA("TextButton")) then return end

            local ok2, allAttrs = pcall(function() return obj:GetAttributes() end)

                    if type(v) == "string" and v:match(UUID_PAT) then return v end
                end

            for _, child in ipairs(obj:GetChildren()) do
                if child:IsA("StringValue") then
                    if (child.Value or ""):match(UUID_PAT) then return child.Value end
                elseif child:IsA("TextLabel") or child:IsA("TextButton") then
                    local v = child.Text or ""
                    if v:match(UUID_PAT) then return v end
                end
