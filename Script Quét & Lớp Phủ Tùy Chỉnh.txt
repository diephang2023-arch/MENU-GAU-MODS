local ENABLE_CUSTOM_OVERLAY = true

local OVERLAY_CONFIG = {
	TEXT = {
		-- ["Chữ gốc"] = "Chữ đổi thành",
	},
	IMAGES = {
		-- ["rbxassetid://123456789"] = "rbxassetid://987654321",

-- =====================================================================
-- 2. BỘ LƯU DỮ LIỆU QUÉT & CHỐNG TRÙNG (TỰ LOẠI TRỪ CHÍNH SCRIPT NÀY)
-- =====================================================================
local SELF_SIGNATURE = "--[[ GAU_SCANNER_OVERLAY_SELF_IGNORE ]]"

local CAPTURED = {
	TEXTS = {},
	IMAGES = {},
	DISCORD = {},
	URLS = {},
	CLIPBOARD = {},
}

local SEEN = {
	TEXTS = {},
	IMAGES = {},
	DISCORD = {},
	URLS = {},
	CLIPBOARD = {},
	SOURCES = {},
}

local onCountUpdated = nil

local function isSelfContent(str)
	if type(str) ~= "string" then
		return false
	end
	return string.find(str, SELF_SIGNATURE, 1, true) ~= nil
		or (string.find(str, "GauScannerOverlayGui", 1, true) ~= nil and string.find(str, "OVERLAY_CONFIG", 1, true) ~= nil)
end

local function totalCapturedCount()
	return #CAPTURED.TEXTS + #CAPTURED.IMAGES + #CAPTURED.DISCORD + #CAPTURED.URLS + #CAPTURED.CLIPBOARD
end

local function notifyCountChanged()
	if type(onCountUpdated) == "function" then
		pcall(onCountUpdated, totalCapturedCount())
	end
end

local function addUniqueItem(category, value, sourceTag)
	if type(value) ~= "string" then
		return
	end
	local trimmed = value:gsub("^%s+", ""):gsub("%s+$", "")
	if trimmed == "" or #trimmed > 500 then
		return
	end
	if isSelfContent(trimmed) then
		return
	end
	if SEEN[category][trimmed] then
		return
	end
	SEEN[category][trimmed] = true

	table.insert(CAPTURED[category], {
		value = trimmed,
		source = sourceTag or "Runtime",
	})
	notifyCountChanged()
end

-- =====================================================================
-- 3. ÁP DỤNG LỚP PHỦ THAY THẾ (CHỈ CHẠY KHI ENABLE_CUSTOM_OVERLAY = true)
-- =====================================================================
local function applyStringOverlay(rawStr)
	if not ENABLE_CUSTOM_OVERLAY or type(rawStr) ~= "string" or rawStr == "" then
		return rawStr
	end

	if OVERLAY_CONFIG.TEXT[rawStr] ~= nil then
		return tostring(OVERLAY_CONFIG.TEXT[rawStr])
	end
	if OVERLAY_CONFIG.LINKS[rawStr] ~= nil then
		return tostring(OVERLAY_CONFIG.LINKS[rawStr])
	end
	if OVERLAY_CONFIG.IMAGES[rawStr] ~= nil then
		return tostring(OVERLAY_CONFIG.IMAGES[rawStr])
	end

	local out = rawStr
	for oldTxt, newTxt in pairs(OVERLAY_CONFIG.TEXT) do
		if type(oldTxt) == "string" and oldTxt ~= "" and string.find(out, oldTxt, 1, true) then
			out = out:gsub(oldTxt:gsub("(%W)", "%%%1"), tostring(newTxt))
		end
	end
	for oldUrl, newUrl in pairs(OVERLAY_CONFIG.LINKS) do
		if type(oldUrl) == "string" and oldUrl ~= "" and string.find(out, oldUrl, 1, true) then
			out = out:gsub(oldUrl:gsub("(%W)", "%%%1"), tostring(newUrl))
		end
	end
	return out
end

local function applyImageOverlay(rawImg)
	if not ENABLE_CUSTOM_OVERLAY or type(rawImg) ~= "string" or rawImg == "" then
		return rawImg
	end

	if OVERLAY_CONFIG.IMAGES[rawImg] ~= nil then
		return tostring(OVERLAY_CONFIG.IMAGES[rawImg])
	end

	local digits = rawImg:match("(%d+)")
	if digits then
		if OVERLAY_CONFIG.IMAGES[digits] ~= nil then
			local rep = tostring(OVERLAY_CONFIG.IMAGES[digits])
			return rep:match("^rbx") and rep or ("rbxassetid://" .. rep)
		end
		local fullKey = "rbxassetid://" .. digits
		if OVERLAY_CONFIG.IMAGES[fullKey] ~= nil then
			local rep = tostring(OVERLAY_CONFIG.IMAGES[fullKey])
			return rep:match("^rbx") and rep or ("rbxassetid://" .. rep)
		end
	end

	return rawImg
end

local function applyClipboardOverlay(rawClip)
	if not ENABLE_CUSTOM_OVERLAY or type(rawClip) ~= "string" or rawClip == "" then
		return rawClip
	end

	if OVERLAY_CONFIG.CLIPBOARD[rawClip] ~= nil then
		return tostring(OVERLAY_CONFIG.CLIPBOARD[rawClip])
	end
	if OVERLAY_CONFIG.LINKS[rawClip] ~= nil then
		return tostring(OVERLAY_CONFIG.LINKS[rawClip])
	end

	local out = rawClip
	for oldVal, newVal in pairs(OVERLAY_CONFIG.CLIPBOARD) do
		if type(oldVal) == "string" and oldVal ~= "" and string.find(out, oldVal, 1, true) then
			out = out:gsub(oldVal:gsub("(%W)", "%%%1"), tostring(newVal))
		end
	end
	for oldUrl, newUrl in pairs(OVERLAY_CONFIG.LINKS) do
		if type(oldUrl) == "string" and oldUrl ~= "" and string.find(out, oldUrl, 1, true) then
			out = out:gsub(oldUrl:gsub("(%W)", "%%%1"), tostring(newUrl))
		end
	end
	return out
end

-- =====================================================================
-- 4. QUÉT NỘI DUNG TẢI VỀ & GIAO DIỆN UI KHI LOAD
-- =====================================================================
local function scanFetchedText(rawStr, sourceTag)
	if type(rawStr) ~= "string" or #rawStr == 0 or isSelfContent(rawStr) then
		return
	end

	local sig = tostring(#rawStr) .. ":" .. rawStr:sub(1, 60) .. ":" .. rawStr:sub(-60)
	if SEEN.SOURCES[sig] then
		return
	end
	SEEN.SOURCES[sig] = true

	for dLink in rawStr:gmatch("https?://[%w%.%-_]*discord[%w%.%-_/%%?&=]+") do
		addUniqueItem("DISCORD", dLink, sourceTag)
	end
	for dShort in rawStr:gmatch("discord%.gg/[%w%-_]+") do
		addUniqueItem("DISCORD", dShort, sourceTag)
	end

	for url in rawStr:gmatch("https?://[%w%.%-_~:/%?#@!$&'%(%)%*%+,;=%%]+") do
		if not url:lower():find("discord", 1, true) then
			addUniqueItem("URLS", url, sourceTag)
		end
	end

	for imgId in rawStr:gmatch("rbx[a-z]+://[%w%d_%-%./%?&=]+") do
		addUniqueItem("IMAGES", imgId, sourceTag)
	end

	for clipStr in rawStr:gmatch('setclipboard%s*%(%s*["\']([^"\'\r\n]+)["\']%s*%)') do
		addUniqueItem("CLIPBOARD", clipStr, sourceTag)
	end
end

local function isOwnGuiInstance(inst)
	local curr = inst
	while curr do
		if curr.Name == "GauScannerOverlayGui" then
			return true
		end
		curr = curr.Parent
	end
	return false
end

local hookedInstances = setmetatable({}, { __mode = "k" })

local function inspectAndHookGuiObject(obj)
	if not obj or isOwnGuiInstance(obj) then
		return
	end

	if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
		local txt = obj.Text
		if type(txt) == "string" and txt ~= "" then
			if txt:lower():find("discord%.gg") or txt:lower():find("discord%.com") then
				addUniqueItem("DISCORD", txt, "UI:" .. obj.ClassName)
			elseif txt:match("^https?://") then
				addUniqueItem("URLS", txt, "UI:" .. obj.ClassName)
			else
				addUniqueItem("TEXTS", txt, "UI:" .. obj.ClassName)
			end

			local replaced = applyStringOverlay(txt)
			if replaced ~= txt then
				obj.Text = replaced
			end
		end

		if obj:IsA("TextBox") and obj.PlaceholderText ~= "" then
			addUniqueItem("TEXTS", obj.PlaceholderText, "UI:Placeholder")
			local repPh = applyStringOverlay(obj.PlaceholderText)
			if repPh ~= obj.PlaceholderText then
				obj.PlaceholderText = repPh
			end
		end

		if not hookedInstances[obj] then
			hookedInstances[obj] = true
			obj:GetPropertyChangedSignal("Text"):Connect(function()
				local newTxt = obj.Text
				if type(newTxt) == "string" and newTxt ~= "" then
					if newTxt:lower():find("discord%.gg") or newTxt:lower():find("discord%.com") then
						addUniqueItem("DISCORD", newTxt, "UI:" .. obj.ClassName)
					elseif newTxt:match("^https?://") then
						addUniqueItem("URLS", newTxt, "UI:" .. obj.ClassName)
					else
						addUniqueItem("TEXTS", newTxt, "UI:" .. obj.ClassName)
					end

					local mapped = applyStringOverlay(newTxt)
					if mapped ~= newTxt then
						obj.Text = mapped
					end
				end
			end)
		end
	end

	if obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
		local img = obj.Image
		if type(img) == "string" and img ~= "" then
			addUniqueItem("IMAGES", img, "UI:" .. obj.ClassName)
			local repImg = applyImageOverlay(img)
			if repImg ~= img then
				obj.Image = repImg
			end
		end

		if not hookedInstances[obj] then
			hookedInstances[obj] = true
			obj:GetPropertyChangedSignal("Image"):Connect(function()
				local newImg = obj.Image
				if type(newImg) == "string" and newImg ~= "" then
					addUniqueItem("IMAGES", newImg, "UI:" .. obj.ClassName)
					local mappedImg = applyImageOverlay(newImg)
					if mappedImg ~= newImg then
						obj.Image = mappedImg
					end
				end
			end)
		end
	end
end

local function attachGuiWatcher(rootContainer)
	if not rootContainer then
		return
	end
	pcall(function()
		for _, desc in ipairs(rootContainer:GetDescendants()) do
			inspectAndHookGuiObject(desc)
		end
		rootContainer.DescendantAdded:Connect(function(desc)
			task.defer(inspectAndHookGuiObject, desc)
		end)
	end)
end

if gethui then
	pcall(function()
		attachGuiWatcher(gethui())
	end)
end
attachGuiWatcher(CoreGui)
if not PlayerGui and LocalPlayer then
	PlayerGui = LocalPlayer:FindFirstChildOfClass("PlayerGui")
end
if PlayerGui then
	attachGuiWatcher(PlayerGui)
end

-- 5. HOOK HTTPGET & SETCLIPBOARD
-- =====================================================================
if hookfunction then
	if game.HttpGet then
		pcall(function()
			local oldHttpGet
			oldHttpGet = hookfunction(game.HttpGet, function(self, url, ...)
				if type(url) == "string" then
					if url:lower():find("discord") then
						addUniqueItem("DISCORD", url, "HttpGet:URL")
					else
						addUniqueItem("URLS", url, "HttpGet:URL")
					end
				end
				local res = oldHttpGet(self, url, ...)
				if type(res) == "string" and #res > 0 and not isSelfContent(res) then
					scanFetchedText(res, "HttpGet:" .. tostring(url))
				end
				return res
			end)
		end)
	end

	local clipTarget = setclipboard or toclipboard
	if type(clipTarget) == "function" then
		pcall(function()
			local oldClip
			oldClip = hookfunction(clipTarget, function(text, ...)
				if type(text) == "string" and text ~= "" and not isSelfContent(text) then
					if text:lower():find("discord%.gg") or text:lower():find("discord%.com") then
						addUniqueItem("DISCORD", text, "ClipboardButton")
					end
					addUniqueItem("CLIPBOARD", text, "ClipboardButton")
					text = applyClipboardOverlay(text)
				end
				return oldClip(text, ...)
			end)
		end)
	end
end

local function escapeLuaStr(s)
	return tostring(s or ""):gsub("\\", "\\\\"):gsub('"', '\\"'):gsub("\r", ""):gsub("\n", "\\n")
end

local function formatEntries(list, mapTable)
	local out = {}
	for _, item in ipairs(list) do
		local chosen = (mapTable and mapTable[item.value] and mapTable[item.value] ~= "") and mapTable[item.value] or item.value
		table.insert(out, string.format('\t\t["%s"] = "%s",', escapeLuaStr(item.value), escapeLuaStr(chosen)))
	end
	return table.concat(out, "\n")
end

local function sanitizeFileName(name)
	local clean = tostring(name or ""):gsub("^%s+", ""):gsub("%s+$", "")
	clean = clean:gsub('[\\/:*?"<>|]', "_")
	return clean
end

local function saveToDeltaFolder(rawName)
	local writeFn = writefile or (syn and syn.write_file) or filewrite
	if not writeFn then
		return false, "Executor không hỗ trợ writefile!"
	end

	local cleanName = sanitizeFileName(rawName)
	if cleanName == "" then
		return false, "Vui lòng đặt tên file trước khi lưu!"
	end

	if not cleanName:lower():match("%.txt$") then
		cleanName = cleanName .. ".txt"
	end

	if type(makefolder) == "function" then
		pcall(function()
			if type(isfolder) == "function" then
				if not isfolder("delta") then
					makefolder("delta")
				end
			else
				makefolder("delta")
			end
		end)
	end

	local fullPath = "delta/" .. cleanName
	local content = buildRunnableOverlayScript()

	local ok, err = pcall(function()
		writeFn(fullPath, content)
	end)

	if ok then
		return true, "Đã lưu script chạy được -> " .. fullPath
	end
	return false, "Lỗi lưu file: " .. tostring(err)
end

	mStroke.Color = Color3.fromRGB(90, 145, 255)
	mStroke.Thickness = 1.5

	local titleLbl = Instance.new("TextLabel")

	titleLbl.Position = UDim2.new(0, 10, 0, 8)
	titleLbl.BackgroundTransparency = 1
	titleLbl.Text = "Lưu vào delta/<tên_file>.txt"
	titleLbl.TextColor3 = Color3.fromRGB(255, 255, 255)
	titleLbl.Font = Enum.Font.GothamBold
	titleLbl.TextSize = 13
	titleLbl.TextXAlignment = Enum.TextXAlignment.Left
	titleLbl.Parent = modal

	local nameBox = Instance.new("TextBox")
	nameBox.Size = UDim2.new(1, -20, 0, 34)
	nameBox.Position = UDim2.new(0, 10, 0, 38)
	nameBox.BackgroundColor3 = Color3.fromRGB(38, 42, 54)
	nameBox.TextColor3 = Color3.fromRGB(255, 255, 255)

	local confirmBtn = Instance.new("TextButton")
	confirmBtn.Size = UDim2.new(0.5, -14, 0, 32)
	confirmBtn.Position = UDim2.new(0.5, 4, 1, -40)
	confirmBtn.BackgroundColor3 = Color3.fromRGB(45, 175, 95)
	confirmBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
	confirmBtn.Font = Enum.Font.GothamBold
	confirmBtn.TextSize = 13
	confirmBtn.Text = "Xác nhận lưu"
	confirmBtn.BorderSizePixel = 0
	confirmBtn.Parent = modal
	Instance.new("UICorner", confirmBtn).CornerRadius = UDim.new(0, 6)

	local dragging, dragStart, startPos, moved = false, nil, nil, false

	floatBtn.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = true
			moved = false
			dragStart = input.Position
			startPos = floatBtn.Position
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement) then
			local delta = input.Position - dragStart
			if delta.Magnitude > 8 then
				moved = true
			end
			floatBtn.Position = UDim2.new(
				startPos.X.Scale,
				startPos.X.Offset + delta.X,
				startPos.Y.Scale,
				startPos.Y.Offset + delta.Y
			)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
			if dragging then
				dragging = false
				if not moved then
					statusLbl.Text = ""
					nameBox.Text = ""
					modal.Visible = not modal.Visible
					if modal.Visible then
						nameBox:CaptureFocus()
					end
				end
			end
		end
	end)

	cancelBtn.MouseButton1Click:Connect(function()
		modal.Visible = false
		statusLbl.Text = ""
	end)

	confirmBtn.MouseButton1Click:Connect(function()
		local ok, msg = saveToDeltaFolder(nameBox.Text)
		if ok then
			statusLbl.TextColor3 = Color3.fromRGB(95, 235, 135)
			statusLbl.Text = msg
			task.delay(0.8, function()
				if modal then
					modal.Visible = false
				end
			end)
		else
			statusLbl.TextColor3 = Color3.fromRGB(255, 110, 110)
			statusLbl.Text = msg
			nameBox:CaptureFocus()
		end
	end)
end
